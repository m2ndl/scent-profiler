  const rated = { opening: null, heart: 1, drydown: null, again: null, chips: {} };
  const ratings = {
    accento: Object.assign({}, rated, { noteAnswers: { white_musk: 1, fruity_sweet: 1 } }),
    lazysunday: Object.assign({}, rated, { noteAnswers: { white_floral: -1 } })

  page.click({ dataset: { none: "1" } });
  const asked = [];
  for (let i = 0; i < 4 && /data-pn=/.test(html(page)); i++) {
    asked.push(...[...html(page).matchAll(/data-pn="([^"]+)" data-pv="0"/g)].map(m => m[1]));

test("zero bottles with sweet and an enjoyed lemon: no pick says the visitor said they avoid a family", () => {
  const answers = { notes: { lemon: 1 }, taste: "sweet" };
  const page = createPage({ localStorage: quizSeed({ pp_quiz_v1: JSON.stringify(answers) }) });
  page.load(scripts);
  const els = page.snapshot().els;
  assert.equal(els["recs-h"].textContent, "Based only on what you told us");
  assert.equal(picksOf(els.recs.innerHTML).length, 3);
  assert.doesNotMatch(els.recs.innerHTML, /which you said you avoid/);
  /* a family pushed down only by the taste answer leans against, in words that do not claim the visitor said it */
  const { picks } = E.recommend(E.computeProfile({ ratings: {}, auto: {}, images: {}, told: N.toldItems(answers) }), {});
  const risk = picks.map(p => p.risks.slice().sort((x, y) => y.sev - x.sev)[0]).find(x => x && x.kind === "told");
  assert.ok(risk, "a pick's main risk comes from the taste answer");
  const stage = { opening: "opening", heart: "heart", drydown: "drydown" }[risk.s];
  assert.match(els.recs.innerHTML, new RegExp(`Contains ${esc(FAMILIES[risk.f].en).replace(/[()]/g, "\\$&")} in the ${stage}, which your quiz answers lean against\\.`));
});
