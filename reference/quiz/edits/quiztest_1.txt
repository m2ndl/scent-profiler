  page.click({ dataset: { taste: o.taste } });

  page.click({ dataset: { continue: "1" } });
  assert.match(html(page), /Do people say a perfume is strong on you when you can barely smell it\?/);
  page.click({ dataset: { anosmia: o.anosmia } });
